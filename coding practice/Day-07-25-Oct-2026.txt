* Ansible - setup  Nginx  with customer config..
  - name: setup Nginx with custom config

   host: all
   become: yes
   tasks: 
         - name: Install Nginx
    yum:
     name: nginx
   status: present
      - name: Copy custom  nginx.conf
   copy:
   src: nginx.conf
   dest: /etc/nginx/nginx.conf
    - name: Start nginx service
    service:
     name: nginx
    state: startd
   enabled: yes
